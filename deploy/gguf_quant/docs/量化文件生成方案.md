## 阶段五：UD 风格量化——自校准动态量化

### 5.1 生成自校准数据（Self-Calibration）

```bash
# 安装 transformers（若未安装）
pip install transformers accelerate

# 执行自校准数据生成脚本
cd /root/autodl-tmp/models/Qwen3.8-27B-quant/scripts

cat > generate_self_calibration.py << 'PYEOF'
#!/usr/bin/env python3
"""
UD 风格自校准数据生成脚本
使用原始 FP16 模型生成校准数据，确保与模型输出分布一致
"""

from transformers import AutoModelForCausalLM, AutoTokenizer
import torch
import os
import sys

MODEL_DIR = "/root/autodl-tmp/models/Qwen3.8-27B-quant/raw_hf_weights"
OUTPUT_FILE = "/root/autodl-tmp/models/Qwen3.8-27B-quant/calibration/self_generated.txt"
NUM_TOPICS = 20
SAMPLES_PER_PROMPT = 5
MAX_NEW_TOKENS = 512

# 多样化任务提示（覆盖技术、代码、分析、设计等场景）
PROMPTS = [
    "请解释深度学习中的注意力机制原理",
    "写一个 Python 函数实现快速排序算法",
    "分析以下代码的时间复杂度和空间复杂度",
    "比较 TCP 和 UDP 协议的优缺点及适用场景",
    "设计一个高并发分布式缓存系统架构",
    "解释量子计算的基本原理和应用前景",
    "编写一个 RESTful API 接口设计规范",
    "分析 Transformer 模型的自注意力机制",
    "如何实现一个高效的向量数据库检索",
    "解释大语言模型中的 KV Cache 机制",
    "编写一个生产者消费者模式的并发代码",
    "分析强化学习中的探索与利用平衡策略",
    "设计一个微服务架构的 API 网关方案",
    "解释图神经网络的消息传递机制",
    "编写一个分布式事务的解决方案",
    "分析数据库索引的 B+ 树实现原理",
    "实现一个简单的词法分析器",
    "解释操作系统中的进程调度算法",
    "设计一个实时流处理系统架构",
    "分析密码学中的 RSA 加密原理",
]

def main():
    print("="*60)
    print("UD 风格自校准数据生成")
    print("="*60)
    
    # 检查 GPU
    if not torch.cuda.is_available():
        print("错误：需要 CUDA GPU")
        sys.exit(1)
    
    print(f"加载模型: {MODEL_DIR}")
    
    # 加载模型和分词器
    model = AutoModelForCausalLM.from_pretrained(
        MODEL_DIR,
        torch_dtype=torch.float16,
        device_map='auto',
        trust_remote_code=True,
        low_cpu_mem_usage=True
    )
    tokenizer = AutoTokenizer.from_pretrained(MODEL_DIR)
    
    print(f"生成 {NUM_TOPICS} 个主题，每个 {SAMPLES_PER_PROMPT} 个样本")
    
    outputs = []
    for i, prompt in enumerate(PROMPTS[:NUM_TOPICS]):
        print(f"[{i+1}/{NUM_TOPICS}] 处理: {prompt[:40]}...")
        
        inputs = tokenizer(prompt, return_tensors='pt').to('cuda')
        
        with torch.no_grad():
            output = model.generate(
                **inputs,
                max_new_tokens=MAX_NEW_TOKENS,
                temperature=0.8,
                top_p=0.95,
                do_sample=True,
                num_return_sequences=SAMPLES_PER_PROMPT,
                pad_token_id=tokenizer.eos_token_id,
                eos_token_id=tokenizer.eos_token_id,
            )
        
        for o in output:
            text = tokenizer.decode(o, skip_special_tokens=True)
            # 清理输出，去除输入提示
            text = text.replace(prompt, '').strip()
            if len(text) > 100:  # 保留足够长度的输出
                outputs.append(text)
    
    # 保存结果
    with open(OUTPUT_FILE, 'w', encoding='utf-8') as f:
        for o in outputs:
            f.write(o + '\n\n')
    
    print("="*60)
    print(f"生成 {len(outputs)} 条自校准数据")
    print(f"保存至: {OUTPUT_FILE}")
    print("="*60)

if __name__ == "__main__":
    main()
PYEOF

chmod +x generate_self_calibration.py

# 执行生成（预计 10-30 分钟，取决于 GPU）
python3 generate_self_calibration.py 2>&1 | tee /root/autodl-tmp/models/Qwen3.8-27B-quant/logs/self_calibration.log
```

```bash
echo "批量生成量化文件..."

for quant in "${QUANTS[@]}"; do
    output="${QUANT_DIR}/Qwen3.8-27B-${quant}.gguf"
    
    if [ -f "${output}" ]; then
        echo "跳过已存在: ${quant}"
        continue
    fi
    
    echo "生成 ${quant}..."
    
    # 判断是否使用 imatrix（低档位推荐用）
    if [[ "${quant}" =~ ^(IQ|Q[0-4]) ]]; then
        ${LLAMA_BIN}/llama-quantize \
            --imatrix ${IMATRIX} \
            ${FP16_MODEL} \
            ${output} \
            ${quant}
    else
        ${LLAMA_BIN}/llama-quantize \
            ${FP16_MODEL} \
            ${output} \
            ${quant}
    fi
    
    echo "完成: ${output}"
done

echo "全部量化完成！"
ls -lh ${QUANT_DIR}/*.gguf
```

## 附录 A：自动化量化转换框架 `quant_framework/`（2026-09-18 重构版）

本地已实现 Python 自动化框架 `quant_framework/`，将本文档全部手工流程封装为可编程流水线。2026-09-18 完成第二次重构，**将单一 pipeline 拆分为两个独立引擎**，职责更清晰、可独立调用。

### A.1 模块架构

```
quant_framework/
├── __init__.py             # 包导出
├── quant_config.py         # 配置：ImatrixMode / QuantType / QuantConfig
├── quant_utils.py          # 共享工具：run_command / ensure_dirs
├── imatrix_generator.py    # 引擎①：imatrix.gguf 生成（4 种模式）
├── gguf_quantizer.py       # 引擎②：FP16→多档位量化 + FP16Converter
├── quant_pipeline.py       # 编排器：串联三引擎 run_all()
└── quant_cli.py            # CLI 入口
```

### A.2 引擎职责边界

| 引擎 | 输入 | 输出 | 可独立调用 |
| :--- | :--- | :--- | :--- |
| `FP16Converter` | safetensors 目录 | `model-fp16.gguf` | ✅ |
| `ImatrixGenerator` | FP16 GGUF + 校准语料 | `imatrix.gguf` 或 `None` | ✅ |
| `GGUFQuantizer` | FP16 GGUF + 可选 imatrix | 多档位量化 GGUF | ✅ |

**关键设计**：`GGUFQuantizer` 在 `imatrix_path=None` 或路径无效时自动降级为标准量化（不附加 `--imatrix` 参数），若目标档位为 IQ 系列则输出质量警告；`--override-tensor` 层级保护与 imatrix 完全解耦。

### A.3 Python API 用法

```python
from quant_framework import QuantConfig, ImatrixMode, QuantType
from quant_framework import ImatrixGenerator, GGUFQuantizer, QuantPipeline

cfg = QuantConfig(
    model_input_dir="/root/autodl-tmp/models/Qwen3.8-27B-FP16",
    output_dir="/root/autodl-tmp/models/quant_output",
    target_quants=[QuantType.Q4_K_M, QuantType.Q5_K_M, QuantType.Q8_0],
    imatrix_mode=ImatrixMode.DATASET,
    calibration_dataset_path="/root/autodl-tmp/calibration.txt",
)

# 方式一：一键流水线
results = QuantPipeline(cfg).run_all()

# 方式二：分步独立调用
gen = ImatrixGenerator(cfg, fp16_gguf_path="model-fp16.gguf", work_dir="./out")
imatrix = gen.generate()                       # NONE 模式返回 None

qz = GGUFQuantizer(cfg, work_dir="./out")
qz.quantize_all("model-fp16.gguf", imatrix_path=imatrix)   # 可传 None 降级
```
